cmake_minimum_required(VERSION 3.24)
project(LinuxBashOS LANGUAGES C)
set(CMAKE_C_STANDARD 17)
include(cmake/Workspace.cmake)
find_package(BpfCapsule CONFIG REQUIRED)
include(cmake/Codegen.cmake)

# Keep allocator and syscall defaults pinned; process termination uses the bridge.
foreach(source allocator.c syscalls.c tlsf_capsule.c)
    configure_file("${BPF_CAPSULE_PLATFORM_DIR}/${source}"
                   "${CMAKE_CURRENT_BINARY_DIR}/platform/${source}" COPYONLY)
endforeach()
configure_file(src/process_exit.c "${CMAKE_CURRENT_BINARY_DIR}/platform/picolibc.c" COPYONLY)
set(BPF_CAPSULE_PLATFORM_DIR "${CMAKE_CURRENT_BINARY_DIR}/platform")

set(BASH_BITCODE_DIRECTORY "${LASHOS_BUILD_DIR}/bitcode" CACHE PATH "GNU Bash guest bitcode")
set(BASH_EXTRA_LIBRARIES "" CACHE STRING "Guest library archives")
option(BASH_USE_ARCHIVES "Link GNU support libraries using their archive inventory" OFF)
set(LINUX_BASH_INLINE_LIMIT 25 CACHE STRING "Capsule inlining limit")
set(LINUX_BASH_FIBER_STACK 2097152 CACHE STRING "Bytes per Capsule fiber stack")
option(LINUX_BASH_VERBOSE "Print Capsule transformation statistics" OFF)
set(capsule_diagnostics)
if(LINUX_BASH_VERBOSE)
    list(APPEND capsule_diagnostics --bpf-capsule-verbose)
endif()
file(GLOB_RECURSE bash_bitcode "${BASH_BITCODE_DIRECTORY}/*.bc")
if(BASH_USE_ARCHIVES)
    list(FILTER bash_bitcode EXCLUDE REGEX "/lib/")
endif()
if(NOT bash_bitcode)
    message(FATAL_ERROR "Run scripts/prepare-bash.py and scripts/compile-bash.py first")
endif()
file(GLOB guest_libc_sources "${CMAKE_CURRENT_SOURCE_DIR}/vendor/musl-c/*.c")
set(guest_libraries)
foreach(archive IN LISTS BASH_EXTRA_LIBRARIES)
    string(MD5 archive_id "${archive}")
    add_library(guest_${archive_id} STATIC IMPORTED)
    set_target_properties(guest_${archive_id} PROPERTIES IMPORTED_LOCATION "${archive}")
    list(APPEND guest_libraries guest_${archive_id})
endforeach()
bpf_capsule_object(bash_bpf OUTPUT bash.bpf.o
    SOURCES src/bash_bpf.c src/bridge_guest.c src/stdio_guest.c src/optional_guest.c src/spawn_guest.c src/memory_guest.c src/services_guest.c ${guest_libc_sources}
    BITCODE ${bash_bitcode}
    LIBRARIES ${guest_libraries}
    INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/include/guest" "${CMAKE_CURRENT_SOURCE_DIR}/include"
    COMPILE_OPTIONS -g -D_GNU_SOURCE -idirafter "${CMAKE_CURRENT_SOURCE_DIR}/vendor/musl-headers"
    LINK_OPTIONS -mcpu=v4 --memory=arena --managed-atomics --allocator-lock=atomic
                 --inline-limit=${LINUX_BASH_INLINE_LIMIT} --native-shift63 --disable-ldsx --freplace --fiber-stack=${LINUX_BASH_FIBER_STACK}
                 ${linux_bash_codegen_options} ${capsule_diagnostics})
bpf_capsule_skeleton(bash_skeleton OUTPUT bash.skel.h OBJECT "${bash_bpf}" NAME bash)
add_executable(linux-bash-os src/host.c src/sandbox_host.c vendor/capsule-host/bpf_capsule_host.c)
target_include_directories(linux-bash-os PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_options(linux-bash-os PRIVATE -Wall -Wextra)
target_link_libraries(linux-bash-os PRIVATE BpfCapsule::host bash_skeleton zstd)
include(CTest)
if(BUILD_TESTING)
    foreach(test queued-signals poll inherited-fds)
        add_executable(${test}-test tests/${test}.c)
        target_include_directories(${test}-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
        target_compile_options(${test}-test PRIVATE -Wall -Wextra -Wno-unused-function -Wno-unused-variable)
        target_link_libraries(${test}-test PRIVATE BpfCapsule::host)
        add_test(NAME ${test} COMMAND ${test}-test)
        set_tests_properties(${test} PROPERTIES TIMEOUT 5)
    endforeach()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(sandbox-test tests/sandbox.c)
    target_compile_options(sandbox-test PRIVATE -UNDEBUG -Wall -Wextra)
    target_link_libraries(sandbox-test PRIVATE zstd)
    add_test(NAME sandbox COMMAND sandbox-test)
    set_tests_properties(sandbox PROPERTIES TIMEOUT 5)
    add_test(NAME source-tree COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/source-tree.py")
    add_test(NAME workspace COMMAND ${Python3_EXECUTABLE} -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/workspace.py")
endif()
install(TARGETS linux-bash-os RUNTIME DESTINATION bin)
install(FILES "${bash_bpf}" DESTINATION bin)
